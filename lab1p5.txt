/* ---------- Data structures ---------- */

typedef struct Node {
    int data;
    struct Node *next;
} Node;

typedef struct {
    Node *head;
    size_t size;
} LinkedList;

/* ---------- Helpers ---------- */

static Node *create_node(int value) {
    Node *node = malloc(sizeof *node);
    if (!node) {
        perror("malloc");
        exit(EXIT_FAILURE);
    }
    node->data = value;
    node->next = NULL;
    return node;
}

void list_init(LinkedList *list) {
    list->head = NULL;
    list->size = 0;
}

/* ---------- Insertion ---------- */

/* Insert at the front: O(1) */
void insert_front(LinkedList *list, int value) {
    Node *node = create_node(value);
    node->next = list->head;
    list->head = node;
    list->size++;
}

/* Insert at the end: O(n) */
void insert_back(LinkedList *list, int value) {
    Node *node = create_node(value);
    if (!list->head) {
        list->head = node;
    } else {
        Node *cur = list->head;
        while (cur->next)
            cur = cur->next;
        cur->next = node;
    }
    list->size++;
}

/* Insert at a zero-based position. Returns 1 on success, 0 if out of range. */
int insert_at(LinkedList *list, size_t index, int value) {
    if (index > list->size)
        return 0;
    if (index == 0) {
        insert_front(list, value);
        return 1;
    }
    Node *prev = list->head;
    for (size_t i = 0; i < index - 1; i++)
        prev = prev->next;
    Node *node = create_node(value);
    node->next = prev->next;
    prev->next = node;
    list->size++;
    return 1;
}

/* ---------- Searching ---------- */

/* Returns zero-based index of first match, or -1 if not found. */
int search(const LinkedList *list, int value) {
    int index = 0;
    for (const Node *cur = list->head; cur; cur = cur->next, index++) {
        if (cur->data == value)
            return index;
    }
    return -1;
}

/* ---------- Deletion ---------- */

/* Delete first node containing value. Returns 1 if deleted, 0 if not found. */

    Node *cur = list->head, *prev = NULL;

        prev = cur;
        cur = cur->next;

    if (!cur)
        return 0;
    if (prev)
        prev->next = cur->next;
    else
        list->head = cur->next;
    free(cur);
    list->size--;
    return 1;
}

/* Delete node at zero-based index. Returns 1 on success, 0 if out of range. */

    if (index >= list->size)
        return 0;
    Node *target;
    if (index == 0) {
        target = list->head;
        list->head = target->next;
    } else {
        Node *prev = list->head;
        for (size_t i = 0; i < index - 1; i++)
            prev = prev->next;
        target = prev->next;
        prev->next = target->next;

    free(target);
    list->size--;

/* ---------- Traversal ---------- */

void print_list(const LinkedList *list) {
    printf("[");
    for (const Node *cur = list->head; cur; cur = cur->next)
        printf("%d%s", cur->data, cur->next ? " -> " : "");
    printf("] (size: %zu)\n", list->size);
}

/* ---------- Cleanup ---------- */

    Node *cur = list->head;
    while (cur) {
        Node *next = cur->next;
        free(cur);
        cur = next;
    }
    list->head = NULL;
    list->size = 0;
}

/* ---------- Data structures ---------- */

    int data;
    struct Node *next;
} Node;

typedef struct {
    Node *head;
    size_t size;
} LinkedList;

static Node *createNode(int value) {
    Node *node = malloc(sizeof *node);
    if (!node) {
        perror("malloc");
        exit(EXIT_FAILURE);
    }
    node->data = value;
    node->next = NULL;
    return node;
}

static void initLinkedList(LinkedList *list) {

    list->size = 0;

/* ---------- Operations ---------- */

/* Insert a new node at the front of the list: O(1) */
void insertAtBeginning(LinkedList *list, int value) {
    Node *node = createNode(value);
    node->next = list->head;
    list->head = node;
    list->size++;
}

/* Insert a new node at the end of the list: O(n) */
void insertAtTheEnd(LinkedList *list, int value) {
    Node *node = createNode(value);
    if (!list->head) {
        list->head = node;
    } else {
        Node *cur = list->head;
        while (cur->next)
            cur = cur->next;
        cur->next = node;
    }
    list->size++;
}

/* Delete the first node holding value. Returns 1 if deleted, 0 if not found. */
int deleteNode(LinkedList *list, int value) {
    Node *cur = list->head, *prev = NULL;
    while (cur && cur->data != value) {
        prev = cur;
        cur = cur->next;
    }
    if (!cur)
        return 0;
    if (prev)
        prev->next = cur->next;
    else
        list->head = cur->next;
    free(cur);
    list->size--;
    return 1;
}

/* Returns the zero-based position of the first node holding value, or -1. */
int findNode(const LinkedList *list, int value) {
    int index = 0;
    for (const Node *cur = list->head; cur; cur = cur->next, index++) {
        if (cur->data == value)
            return index;
    }
    return -1;
}

/* Print every node from head to tail. */
void displayLinkedList(const LinkedList *list) {
    if (!list->head) {
        printf("List is empty (size: 0)\n");
        return;

    for (const Node *cur = list->head; cur; cur = cur->next)
        printf("%d -> ", cur->data);
    printf("NULL (size: %zu)\n", list->size);
}

static void freeLinkedList(LinkedList *list) {

/* Parse a strict base-10 int. Returns 1 on success, 0 on failure. */
static int parseInt(const char *s, int *out) {
    char *end;
    errno = 0;
    long v = strtol(s, &end, 10);
    if (end == s || *end != '\0' || errno == ERANGE || v < INT_MIN || v > INT_MAX)

    *out = (int)v;
